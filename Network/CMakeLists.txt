cmake_minimum_required(VERSION 3.24)
project(OpenRuneNet LANGUAGES C CXX)
include(CTest)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(OPENRUNE_NET_SANITIZE "Enable address/undefined behavior sanitizers" OFF)
option(OPENRUNE_NET_FUZZ "Build libFuzzer packet decoder target (Clang)" OFF)
include(FetchContent)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
FetchContent_Declare(mbedtls
  URL https://github.com/Mbed-TLS/mbedtls/releases/download/mbedtls-3.6.7/mbedtls-3.6.7.tar.bz2
  URL_HASH SHA256=a7e8bcbec0e6f761b4af24f25677626b35f762f68eef79c08677a363212d11f6
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(mbedtls)
file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/thirdparty-notices/" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/network-notices")
add_library(openrune_net STATIC src/Protocol.cpp src/Transport.cpp src/ClockSync.cpp src/SecureChannel.cpp src/SecureSession.cpp src/Replication.cpp src/Simulation.cpp src/Prediction.cpp src/Presentation.cpp src/UdpSocket.cpp)
if(WIN32)
  target_link_libraries(openrune_net PUBLIC ws2_32)
endif()
target_link_libraries(openrune_net PUBLIC mbedtls mbedcrypto)
find_package(Threads REQUIRED)
target_link_libraries(openrune_net PUBLIC Threads::Threads)
target_include_directories(openrune_net PUBLIC include)
target_compile_features(openrune_net PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(openrune_net PRIVATE /W4 /WX)
else()
  target_compile_options(openrune_net PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(OPENRUNE_NET_SANITIZE)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "OPENRUNE_NET_SANITIZE requires Clang or GCC")
  endif()
  target_compile_options(openrune_net PUBLIC -O1 -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(openrune_net PUBLIC -fsanitize=address,undefined)
  foreach(dependency mbedtls mbedx509 mbedcrypto)
    target_compile_options(${dependency} PRIVATE -O1 -fsanitize=address,undefined -fno-omit-frame-pointer)
  endforeach()
endif()
if(BUILD_TESTING)
  add_executable(net_presentation_tests tests/PresentationTests.cpp)
  target_link_libraries(net_presentation_tests PRIVATE openrune_net)
  add_test(NAME net_presentation COMMAND net_presentation_tests)
  add_executable(net_diagnostics_tests tests/DiagnosticsTests.cpp)
  target_link_libraries(net_diagnostics_tests PRIVATE openrune_net)
  add_test(NAME net_diagnostics COMMAND net_diagnostics_tests)
  add_executable(net_udp_socket_tests tests/UdpSocketTests.cpp)
  target_link_libraries(net_udp_socket_tests PRIVATE openrune_net)
  add_test(NAME net_production_udp COMMAND net_udp_socket_tests)
  set_tests_properties(net_production_udp PROPERTIES TIMEOUT 15)
  add_executable(net_melee_timing_tests tests/MeleeTimingTests.cpp)
  target_link_libraries(net_melee_timing_tests PRIVATE openrune_net)
  add_test(NAME net_melee_timing COMMAND net_melee_timing_tests)
  add_executable(net_prediction_tests tests/PredictionTests.cpp)
  target_link_libraries(net_prediction_tests PRIVATE openrune_net)
  add_test(NAME net_prediction COMMAND net_prediction_tests)
  add_executable(net_simulation_tests tests/SimulationTests.cpp)
  target_link_libraries(net_simulation_tests PRIVATE openrune_net)
  add_test(NAME net_simulation COMMAND net_simulation_tests)
  add_executable(net_replication_tests tests/ReplicationTests.cpp)
  target_link_libraries(net_replication_tests PRIVATE openrune_net)
  add_test(NAME net_replication COMMAND net_replication_tests)
  add_executable(net_tests tests/NetTests.cpp)
  target_link_libraries(net_tests PRIVATE openrune_net)
  add_executable(net_security_tests tests/SecurityTests.cpp)
  target_link_libraries(net_security_tests PRIVATE openrune_net)
  add_test(NAME net_authentication_clocks COMMAND net_security_tests)
  add_executable(net_udp_peer tests/UdpPeer.cpp)
  find_package(Threads REQUIRED)
  target_link_libraries(net_udp_peer PRIVATE openrune_net Threads::Threads)
  if(WIN32)
    target_link_libraries(net_udp_peer PRIVATE ws2_32)
  endif()
  add_test(NAME net_contract COMMAND net_tests contract)
  add_test(NAME net_impairments COMMAND net_tests impairments)
  add_test(NAME net_soak COMMAND net_tests soak)
  set_tests_properties(net_soak PROPERTIES TIMEOUT 120)
  if(NOT CMAKE_CROSSCOMPILING)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME net_replication_wire COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/replication_wire.py $<TARGET_FILE:net_replication_tests>)
    add_test(NAME net_independent_wire COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/wire_reference.py $<TARGET_FILE:net_tests>)
    add_test(NAME net_udp_loopback COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/udp_loopback.py $<TARGET_FILE:net_udp_peer>)
    set_tests_properties(net_udp_loopback PROPERTIES TIMEOUT 30)
    add_test(NAME net_dtls_loopback COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/udp_loopback.py $<TARGET_FILE:net_udp_peer> secure)
    set_tests_properties(net_dtls_loopback PROPERTIES TIMEOUT 30)
  endif()
endif()
if(OPENRUNE_NET_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT OPENRUNE_NET_SANITIZE)
    message(FATAL_ERROR "OPENRUNE_NET_FUZZ requires Clang and OPENRUNE_NET_SANITIZE=ON")
  endif()
  target_compile_options(openrune_net PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
  add_executable(net_fuzz tests/FuzzDecode.cpp)
  target_compile_options(net_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(net_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_libraries(net_fuzz PRIVATE openrune_net)
  foreach(dependency mbedtls mbedx509 mbedcrypto)
    target_compile_options(${dependency} PRIVATE -fsanitize=fuzzer-no-link)
  endforeach()
  add_executable(net_replication_fuzz tests/FuzzReplication.cpp)
  target_compile_options(net_replication_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(net_replication_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_libraries(net_replication_fuzz PRIVATE openrune_net)
  add_executable(net_security_fuzz tests/FuzzSecurity.cpp)
  target_compile_options(net_security_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(net_security_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_libraries(net_security_fuzz PRIVATE openrune_net)
endif()
